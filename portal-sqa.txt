    <script>
        let dadosOriginais = [];
        let uploadTempData = {}; 

        google.script.run
            .withSuccessHandler(function(respostaString) {
                try {
                    let resposta = JSON.parse(respostaString);
                    if(resposta.erroCritico) {
                        document.getElementById('corpo-tabela').innerHTML = `<tr><td colspan="4" style="text-align: center; color: var(--danger); font-weight: bold; padding: 40px;">⚠️ Erro: ${resposta.erroCritico}</td></tr>`;
                        return;
                    }
                    document.getElementById('link-planilha').href = resposta.urlPlanilha;
                    dadosOriginais = resposta.fornecedores;
                    
                    dadosOriginais.sort((a, b) => a.score - b.score);
                    renderizarTabela(dadosOriginais);
                } catch(e) {
                    document.getElementById('corpo-tabela').innerHTML = `<tr><td colspan="4" style="text-align: center; color: var(--danger); padding: 40px;">⚠️ Erro: ${e.message}</td></tr>`;
                }
            })
            .pegarDadosDetalhados();

        function renderizarTabela(dados) {
            const tabela = document.getElementById('corpo-tabela');
            tabela.innerHTML = '';

            if(dados.length === 0) {
                tabela.innerHTML = `<tr><td colspan="4" style="text-align: center; padding: 40px; color: #888;">Nenhum fornecedor encontrado com a combinação de filtros atual.</td></tr>`;
                return;
            }

            dados.forEach(f => {
                let corScore = f.score >= 100 ? 'var(--success)' : (f.score >= 50 ? '#f39c12' : 'var(--danger)');

                let linhaHTML = `
                    <tr class="linha-fornecedor" onclick="toggleDetalhe('${f.id}')">
                        <td><strong>${f.id}</strong></td>
                        <td>${f.nome}</td>
                        <td>
                            <div class="score-container">
                                <span class="score-text">${f.score}%</span>
                                <div class="score-bg"><div class="score-fill" style="width: ${f.score}%; background-color: ${corScore};"></div></div>
                            </div>
                        </td>
                        <td style="text-align: right; color: var(--blue); font-size: 13px; font-weight: 600;" id="icon-${f.id}">⮟ Expandir</td>
                    </tr>
                `;

                let docsHTML = f.documentos.map(d => {
                    let botoes = '';
                    if (d.link) {
                        botoes = `<a href="${d.link}" target="_blank" class="btn-link">Ver Arquivo ↗</a>`;
                    } else if (d.linhaBase === -1) { 
                        botoes = `<span style="font-size: 11px; color: #f39c12; font-style: italic;">Não localizado</span>`;
                    } else if (d.cor === '#e74c3c') { 
                        botoes = `<button class="btn-upload" onclick="abrirSeletor('${f.id}', '${f.nome}', '${d.abaDestino}', '${d.linhaBase}', '${d.colStatus}'); event.stopPropagation();">↑ Anexar PDF</button>`;
                    } else {
                        botoes = `<span style="font-size: 11px; color: #999; font-style: italic;">Não Aplicável</span>`;
                    }

                    return `
                        <div class="doc-card" style="border-left-color: ${d.cor};">
                            <h4>${d.nome}</h4>
                            <p style="color: ${d.cor === '#e74c3c' || d.cor === '#f39c12' ? d.cor : '#777'}; font-weight: ${d.cor === '#e74c3c' || d.cor === '#f39c12' ? 'bold' : 'normal'};">Status: ${d.status}</p>
                            ${botoes}
                        </div>
                    `;
                }).join('');

                linhaHTML += `
                    <tr id="detalhe-${f.id}" class="linha-detalhes">
                        <td colspan="4" style="padding: 0; border-bottom: 2px solid #ddd;">
                            <div class="tabs-nav">
                                <button class="tab-btn active" onclick="mudarAba(event, 'docs-${f.id}')">📄 Auditoria e Certificados</button>
                                <button class="tab-btn" onclick="mudarAba(event, 'info-${f.id}')">ℹ️ Ficha Cadastral</button>
                            </div>
                            <div id="docs-${f.id}" class="tab-content active"><div class="docs-grid">${docsHTML}</div></div>
                            <div id="info-${f.id}" class="tab-content">
                                <div class="ficha-fornecedor">
                                    <div class="ficha-item"><span>País de Origem</span><strong>${f.pais}</strong></div>
                                    <div class="ficha-item"><span>Status ERP</span><strong>${f.statusGeral}</strong></div>
                                    <div class="ficha-item" style="width: 100%;"><span>Contatos (E-mail)</span><strong>${f.emails}</strong></div>
                                </div>
                            </div>
                        </td>
                    </tr>
                `;
                tabela.innerHTML += linhaHTML;
            });
        }

        window.toggleDetalhe = function(id) {
            const linha = document.getElementById(`detalhe-${id}`);
            const icone = document.getElementById(`icon-${id}`);
            if (linha.style.display === 'table-row') {
                linha.style.display = 'none'; icone.innerText = '⮟ Expandir';
            } else {
                linha.style.display = 'table-row'; icone.innerText = '⮝ Fechar';
            }
        };

        window.mudarAba = function(evento, idAba) {
            evento.stopPropagation(); 
            let linhaDetalhe = evento.target.closest('.linha-detalhes');
            linhaDetalhe.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            linhaDetalhe.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
            evento.target.classList.add('active');
            document.getElementById(idAba).classList.add('active');
        };

        window.filtrarDados = function() {
            const busca = document.getElementById('busca').value.toLowerCase();
            const statusFiltro = document.getElementById('filtro-status').value;
            const progressoFiltro = document.getElementById('filtro-progresso').value;

            const filtrados = dadosOriginais.filter(f => {
                const confereBusca = f.nome.toLowerCase().includes(busca) || f.id.toLowerCase().includes(busca);
                
                let confereStatus = true;
                const statusGeral = f.statusGeral ? f.statusGeral.toUpperCase() : '';
                if (statusFiltro === 'OK') {
                    confereStatus = statusGeral.includes('OK') && !statusGeral.includes('NOK');
                } else if (statusFiltro === 'NOK') {
                    confereStatus = statusGeral.includes('NOK') || statusGeral.includes('PENDENTE') || statusGeral.includes('ATRASADO') || statusGeral.includes('ANALISAR');
                }

                let confereProgresso = true;
                if (progressoFiltro === '100') {
                    confereProgresso = (f.score === 100);
                } else if (progressoFiltro === 'parcial') {
                    confereProgresso = (f.score > 0 && f.score < 100);
                } else if (progressoFiltro === '0') {
                    confereProgresso = (f.score === 0);
                }

                return confereBusca && confereStatus && confereProgresso;
            });

            renderizarTabela(filtrados);
        };

        window.abrirSeletor = function(idFornecedor, nomeFornecedor, abaDestino, linhaBase, colStatus) {
            uploadTempData = { idFornecedor, nomeFornecedor, abaDestino, linhaBase, colStatus };
            document.getElementById('arquivo-upload').click();
        };

        window.processarUpload = function(input) {
            const arquivo = input.files[0];
            if (!arquivo) return;

            document.getElementById('loading-text').innerText = 'A guardar e atualizar o sistema...';
            document.getElementById('loading-overlay').style.display = 'flex';
            
            const reader = new FileReader();
            reader.onload = function(e) {
                const base64Data = e.target.result.split(',')[1];
                const pacoteUpload = { nomeArquivo: arquivo.name, mimeType: arquivo.type, base64: base64Data, ...uploadTempData };
                
                google.script.run.withSuccessHandler(function(res) {
                    document.getElementById('loading-overlay').style.display = 'none';
                    if(res.sucesso) { 
                        alert('Arquivo anexado! A planilha foi atualizada.'); 
                        location.reload(); 
                    } else { 
                        alert('Erro ao enviar: ' + res.erro); 
                    }
                }).receberUpload(pacoteUpload);
            };
            reader.readAsDataURL(arquivo);
        };
    </script>
